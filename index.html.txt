<script>

function cekSegitiga() {

    const a = Number(document.getElementById("a").value);
    const b = Number(document.getElementById("b").value);
    const c = Number(document.getElementById("c").value);

    const hasil = document.getElementById("hasil");

    if (!a || !b || !c) {
        hasil.innerHTML = "⚠️ Data belum lengkap!";
        return;
    }

    if (a + b <= c || a + c <= b || b + c <= a) {
        hasil.innerHTML = "❌ Sisi tidak dapat membentuk segitiga.";
        return;
    }

    if (
        a ** 2 + b ** 2 === c ** 2 ||
        a ** 2 + c ** 2 === b ** 2 ||
        b ** 2 + c ** 2 === a ** 2
    ) {
        hasil.innerHTML = "📐 Jenis: Segitiga Siku-Siku";
    }

    else if (a === b || a === c || b === c) {
        hasil.innerHTML = "🔺 Jenis: Segitiga Sama Kaki";
    }

    else {
        hasil.innerHTML = "🔻 Jenis: Segitiga Sembarang";
    }
}

</script>
